27__global__ void raytrace(uchar4 * __restrict__ buffer, const int width, const int height, const Camera cam,
28 const Vec3 bg, const Light light, const uint32_t frameSeed) {
29 // Thread → pixel
30 const unsigned int x = blockIdx.x * blockDim.x + threadIdx.x;
31 const unsigned int y = blockIdx.y * blockDim.y + threadIdx.y;
32 if (x >= static_cast<unsigned int>(width) ||
33 y >= static_cast<unsigned int>(height)) {
34 return;
35 }
36 const int idx = static_cast<int>(y) * width + static_cast<int>(x);
37
38 // Primary ray
39 const Ray ray = generatePrimaryRay(cam, x, y, width, height);
40
41#if DEBUG_DRAW_LIGHT_SPHERE || DEBUG_DRAW_LIGHT_DIRECTION
42 // Debug gizmos on top (early-out on hit).
43 {
44 uchar3 gizmoColor;
45 if (renderLightDebug(ray, light, gizmoColor)) {
46 buffer[idx] = make_uchar4(gizmoColor.x, gizmoColor.y, gizmoColor.z, 255);
47 return;
48 }
49 if (renderLightDirectionRay(ray, light, gizmoColor)) {
50 buffer[idx] = make_uchar4(gizmoColor.x, gizmoColor.y, gizmoColor.z, 255);
51 return;
52 }
53 }
54#endif
55
56 // Device scene view
57 const SceneGeom G = getDeviceScene();
58
59 // Closest hit
60 Hit h{};
61 traceClosest(ray, G, h);
62
63 // Shading (unified with CPU path)
64 const int softSamples = defaultUseSoftShadows() ? defaultSoftShadowSamples() : 0; // 0 = hard
65 const bool useBent = defaultUseBentShadows();
66 constexpr int maxDepth = 4;
67
68 // Per-pixel RNG seed (scramble with pixel coords)
69 uint32_t seed = frameSeed;
70 seed ^= 0x9E3779B1u * (static_cast<uint32_t>(x) + 1u);
71 seed ^= 0x85EBCA77u * (static_cast<uint32_t>(y) + 1u);
72 seed ^= 0xC2B2AE3Du;
73
74 // shadeSurface returns gamma-encoded linear RGB in [0,1]
75 const Vec3 linearGamma = shadeSurface(
76 h, ray, light, G,
77 seed, maxDepth, softSamples,
78 bg, useBent
79 );
80
81 const uchar3 rgb = toUChar3(linearGamma);
82 buffer[idx] = make_uchar4(rgb.x, rgb.y, rgb.z, 255);
83}

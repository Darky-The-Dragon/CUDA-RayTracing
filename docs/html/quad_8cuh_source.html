65 HD FINL inline bool intersect(const Ray &ray, float &tHit) const {
66 // pVec = D × spanV
67 const Vec3 pVec = ray.direction.cross(spanV);
68
69 // det = spanU · pVec — parallel if ~ 0
70 const float det = spanU.dot(pVec);
71 if (fabsf(det) < num::kEps()) return false;
72
73 const float invDet = 1.0f / det;
74
75 // displacement from corner to ray origin
76 const Vec3 s = ray.origin - position;
77
78 // u parameter along spanU
79 const float u = s.dot(pVec) * invDet;
80 if (u < 0.0f || u > 1.0f) return false;
81
82 // qVec = s × spanU
83 const Vec3 qVec = s.cross(spanU);
84
85 // v parameter along spanV
86 const float v = ray.direction.dot(qVec) * invDet;
87 if (v < 0.0f || v > 1.0f) return false;
88
89 // ray distance
90 const float t = spanV.dot(qVec) * invDet;
91 if (t > num::kEps()) {
92 tHit = t;
93 return true;
94 }

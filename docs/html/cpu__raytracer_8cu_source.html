34__host__ void cpu_raytrace(uchar3 *buffer, int width, int height, uint32_t sceneMask, const DebugConfigHost &dbg,
35 const uint32_t frameSeed) {
36 // ------------------------
37 // Scene setup (from mask)
38 // ------------------------
39 WorldBuffers W;
40 buildWorld(W, sceneMask); // host-only world composition
41 SceneGeom G{
42 W.quads, W.numQuads,
43 W.spheres, W.numSpheres
44 };
45
46 // ------------------------
47 // Environment & lighting
48 // ------------------------
49 Camera cam;
50 cam.fov_deg = defaultCameraFovDeg();
51
52 const Vec3 bg = toFloat3(defaultBackgroundU8()); // background when no hit
53 const Light light = defaultLight(); // shared default light
54
55 // ------------------------
56 // Per-pixel rendering loop
57 // ------------------------
58 for (int y = 0; y < height; ++y) {
59 for (int x = 0; x < width; ++x) {
60 const int idx = y * width + x;
61
62 // Ray generation
63 const Ray ray = generatePrimaryRay(cam, x, y, width, height);
64
65 // Debug gizmos (draw on top), same as GPU path. Early-out on hit.
66#if DEBUG_DRAW_LIGHT_SPHERE
67 if (dbg.drawLightSphere) {
68 if (uchar3 gizmoColor; renderLightDebug(ray, light, gizmoColor)) {
69 buffer[idx] = gizmoColor;
70 continue;
71 }
72 }
73#endif
74#if DEBUG_DRAW_LIGHT_DIRECTION
75 if (dbg.drawLightDir) {
76 if (uchar3 gizmoColor; renderLightDirectionRay(ray, light, gizmoColor)) {
77 buffer[idx] = gizmoColor;
78 continue;
79 }
80 }
81#endif
82#if DEBUG_DRAW_NORMALS
83 // If/when a normal-visualization gizmo lands here, gate with dbg.drawNormals.
84#endif
85
86 // Closest hit
87 Hit hit{};
88 traceClosest(ray, G, hit);
89
90 // Shading (unified CPU/GPU)
91 const int softSamples = defaultUseSoftShadows() ? defaultSoftShadowSamples() : 0; // 0 = hard
92 const bool useBent = defaultUseBentShadows();
93 constexpr int maxDepth = 4; // primary + a couple of bounces
94
95 // Per-pixel RNG seed (scramble with pixel coords)
96 uint32_t seed = frameSeed;
97 seed ^= 0x9E3779B1u * (static_cast<uint32_t>(x) + 1u);
98 seed ^= 0x85EBCA77u * (static_cast<uint32_t>(y) + 1u);
99 seed ^= 0xC2B2AE3Du;
100
101 // shadeSurface returns gamma-encoded color in [0,1]
102 const Vec3 color = shadeSurface(
103 hit, ray, light, G,
104 seed, maxDepth, softSamples,
105 bg, useBent
106 );
107
108 buffer[idx] = toUChar3(color);
109 }
110 }
111}

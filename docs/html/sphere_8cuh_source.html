28HD FINL inline bool sphereIntersectRoots(const Vec3 &O, const Vec3 &D, const Vec3 &C, const float r, float &t0,
29 float &t1) {
30 const Vec3 oc = O - C;
31 const float a = D.dot(D); // = 1 if D normalized
32 const float b = 2.0f * oc.dot(D);
33 const float c = oc.dot(oc) - r * r;
34 const float disc = b * b - 4.f * a * c;
35 if (disc < 0.0f) return false;
36
37 const float inv2a = 0.5f / a;
38 const float s = sqrtf(disc);
39 float tn = (-b - s) * inv2a;
40 float tf = (-b + s) * inv2a;
41 if (tn > tf) {
42 const float tmp = tn;
43 tn = tf;
44 tf = tmp;
45 }
46
47 t0 = tn;
48 t1 = tf;
49 return true;
50}

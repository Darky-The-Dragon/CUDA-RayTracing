50static bool writePPM_P6(const std::string &path, const uchar3 *pixels, const int w, const int h) {
51 if (!pixels || w <= 0 || h <= 0) return false;
52
53 std::ofstream out(path, std::ios::binary);
54 if (!out) return false;
55
56 out << "P6\n" << w << ' ' << h << "\n255\n";
57
58 const auto total = static_cast<size_t>(w) * static_cast<size_t>(h) * sizeof(uchar3);
59 if (total > static_cast<size_t>(std::numeric_limits<std::streamsize>::max())) return false;
60
61 const auto count = static_cast<std::streamsize>(total);
62 out.write(reinterpret_cast<const char *>(pixels), count);
63 return static_cast<bool>(out);
64}

96bool saveImage(const std::string &path, const uchar3 *pixels, const int w, const int h, const ExportFormat fmt) {
97 if (!pixels || w <= 0 || h <= 0 || path.empty()) return false;
98
99 // Create output directory once here (writers don’t repeat it).
100 std::error_code ec;
101 fs::create_directories(fs::path(path).parent_path(), ec);
102
103 switch (fmt) {
104 case ExportFormat::PPM: return writePPM_P6(path, pixels, w, h);
105 case ExportFormat::PNG: return writePNG(path, pixels, w, h);
106 default: return false;
107 }

231void addWatermarkInPlace(std::vector<uchar3> &img, const int w, const int h, const std::string &text) {
232 if (text.empty() || w <= 0 || h <= 0) return;
233
234 // Uppercase to fit glyph set.
235 std::string textIn;
236 textIn.reserve(text.size());
237 for (const char c: text)
238 textIn.push_back(static_cast<char>(std::toupper(static_cast<unsigned char>(c))));
239
240 // Glyph metrics (5×7 + 1px spacing). Add a small pad around the label.
241 constexpr int gw = 5, gh = 7, gap = 1;
242 constexpr int margin = 6;
243 const int text_w = static_cast<int>(textIn.size()) * (gw + gap) - gap;
244 constexpr int text_h = gh;
245
246 int x0 = w - text_w - margin;
247 int y0 = h - text_h - margin;
248 x0 = std::max(0, x0);
249 y0 = std::max(0, y0);
250
251 // Darken backdrop rectangle (expand by 2 px).
252 const int bx0 = std::max(0, x0 - 2);
253 const int by0 = std::max(0, y0 - 2);
254 const int bx1 = std::min(w - 1, x0 + text_w + 2);
255 const int by1 = std::min(h - 1, y0 + text_h + 2);
256 for (int y = by0; y <= by1; ++y) {
257 auto *row = img.data() + static_cast<size_t>(y) * w;
258 for (int x = bx0; x <= bx1; ++x) {
259 uchar3 &p = row[x];
260 p.x = darken30(p.x);
261 p.y = darken30(p.y);
262 p.z = darken30(p.z);
263 }
264 }
265
266 // Draw glyphs in opaque white.
267 int penX = x0;
268 for (const char c: textIn) {
269 const Glyph *g = glyphFor(c);
270 for (int ry = 0; ry < gh; ++ry) {
271 const uint8_t row = g->rows[ry];
272 for (int rx = 0; rx < gw; ++rx) {
273 if (row & (1u << (gw - 1 - rx))) {
274 putPixel(img, w, h, penX + rx, y0 + ry, make_uchar3(255, 255, 255));
275 }
276 }
277 }
278 penX += gw + gap;
279 }
280}

295 inline std::wstring toAbsoluteWide(const std::string &p) {
296 // Absolutize (swallow exceptions).
297 std::string abs;
298 try {
299 abs = fs::absolute(p).string();
300 } catch (...) {
301 abs = p;
302 }
303
304 if (abs.empty()) return {};
305 const int need = ::MultiByteToWideChar(CP_UTF8, 0, abs.c_str(),
306 static_cast<int>(abs.size()), nullptr, 0);
307 if (need <= 0) {
308 // Fallback: ASCII-safe widening.
309 return {abs.begin(), abs.end()};
310 }
311 std::wstring out;
312 out.resize(static_cast<size_t>(need));
313 ::MultiByteToWideChar(CP_UTF8, 0, abs.c_str(),
314 static_cast<int>(abs.size()), out.data(), need);
315 return out;
316 }

341bool openPreview(const std::string &path) {
342 if (!fs::exists(fs::path(path))) {
343 std::cerr << "[PREVIEW] File does not exist: " << path << "\n";
344 return false;
345 }
346
347 const std::wstring wabs = toAbsoluteWide(path);
348
349#if RT_PREVIEW_WINDOWS
350 HINSTANCE r = ::ShellExecuteW(nullptr, L"open", wabs.c_str(), nullptr, nullptr, SW_SHOWNORMAL);
351 if (const auto code = reinterpret_cast<intptr_t>(r); code <= 32) {
352 std::string abs;
353 try { abs = fs::absolute(path).string(); } catch (...) { abs = path; }
354 std::cerr << "[PREVIEW] ShellExecuteW failed (code=" << code
355 << "): " << shellCodeMeaning(code) << " | Path: " << abs << "\n";
356 return false;
357 }
358#endif
359
360 std::cout << "[PREVIEW] Opened: " << path << "\n";
361 return true;

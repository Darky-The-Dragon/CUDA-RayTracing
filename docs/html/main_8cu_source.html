107static void chooseLaunchDimsRaytrace(const int width, const int height, dim3 &grid, dim3 &block) {
108 int minGridSize = 0, optBlockSize = 0;
109 CUDA_GUARD(cudaOccupancyMaxPotentialBlockSize(
110 &minGridSize, &optBlockSize, raytrace, /*dynSMem=*/0, /*limit=*/0));
111
112 if (optBlockSize <= 0) {
113 block = dim3(16, 16, 1);
114 } else {
115 int bx = 16, by = std::max(1, optBlockSize / bx);
116 while (bx * by > optBlockSize && by > 1) by >>= 1;
117 if (bx * by > optBlockSize) {
118 bx = optBlockSize;
119 by = 1;
120 }
121 block = dim3(bx, by, 1);
122 }
123 grid = dim3((width + block.x - 1) / block.x,
124 (height + block.y - 1) / block.y,
125 1);
126
127 RT_DEBUG_ONLY(std::cout
128 << "[LAUNCH] Occupancy-picked block: (" << block.x << "x" << block.y << ")\n"
129 << "[LAUNCH] Grid: (" << grid.x << "x" << grid.y << ")\n");
130}

141static std::string makeSceneLabel(int mask) {
142 std::string s;
143 const auto m = static_cast<std::uint32_t>(mask);
144 auto append = [&](const char *name) {
145 if (!s.empty()) s += " | ";
146 s += name;
147 };
148 if (m & SCENE_CORNELL) append("Cornell");
149 if (m & SCENE_SPHERES) append("Spheres");
150 if (m & SCENE_CUBES) append("Cubes");
151 if (s.empty()) s = "None";
152 return s;
153}

184static bool loadLastCRC(const fs::path &cachePath, const std::string &key, uint32_t &out) {
185 std::ifstream in(cachePath);
186 if (!in) return false;
187 std::string line;
188 bool found = false;
189 while (std::getline(in, line)) {
190 auto comma = line.find(',');
191 if (comma == std::string::npos) continue;
192 if (line.compare(0, comma, key) == 0) {
193 // take the last occurrence
194 std::string val = line.substr(comma + 1);
195 try {
196 out = static_cast<uint32_t>(std::stoul(val));
197 found = true;
198 } catch (...) {
199 /* ignore parse errors */
200 }
201 }
202 }
203 return found;
204}

212static void appendCRC(const fs::path &cachePath, const std::string &key, uint32_t crc) {
213 fs::create_directories(cachePath.parent_path());

268 // Strip alpha (RGBA->RGB) on host
269 auto stripAlphaToRGB = [&](const uchar4 *src, uchar3 *dst, size_t pixels) {
270 for (size_t i = 0; i < pixels; ++i) {
271 const uchar4 p = src[i];
272 dst[i] = make_uchar3(p.x, p.y, p.z);
273 }

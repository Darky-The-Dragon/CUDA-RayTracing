146HD FINL inline Vec3 normalized(const Vec3 &v) {
147 const float l2 = lengthSquared(v);
148 if (l2 <= 0.0f) return Vec3(0.0f);
149#ifdef __CUDA_ARCH__
150 const float inv = rsqrtf(l2); // fast path on GPU
151 return Vec3{v.x * inv, v.y * inv, v.z * inv};
152#else
153 const float l = std::sqrt(l2);
154 const float inv = (l > 0.f) ? (1.f / l) : 0.f;
155 return Vec3{v.x * inv, v.y * inv, v.z * inv};
156#endif

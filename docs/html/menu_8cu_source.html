41 template<class T, class ClampFn>
42 T readNumberOrDefault(const std::string &prompt, T def, ClampFn clamp) {
43 if (!prompt.empty()) {
44 std::cout << prompt << " [" << def << "]: ";
45 }
46 std::string line;
47 std::getline(std::cin, line);
48
49 // Trim whitespace
50 const auto l = line.find_first_not_of(" \t\r\n");
51 if (l == std::string::npos) return def;
52 const auto r = line.find_last_not_of(" \t\r\n");
53 line = line.substr(l, r - l + 1);
54
55 if (line.empty()) return def;
56
57 try {
58 if constexpr (std::is_floating_point_v<T>) {
59 T v = static_cast<T>(std::stod(line));
60 return clamp(v);
61 } else {
62 long long v = std::stoll(line);
63 return clamp(static_cast<T>(v));
64 }
65 } catch (...) {
66 return def;
67 }
68 }

99 std::string sceneMaskToString(const int mask) {

103 if (m & SCENE_CORNELL) {

111 if (m & SCENE_CUBES) {
112 s += (s.empty() ? "" : " | ");

115

139 void submenuResolution(RuntimeConfig &rc) {
140 for (;;) {
141 std::cout << "\n== Resolution ==\n";
142 std::cout << "Current: " << rc.width << " x " << rc.height << "\n";
143 std::cout << "Common presets:\n";
144 for (size_t i = 0; i < std::size(kCommonRes); ++i) {
145 std::cout << " (" << (i + 1) << ") "
146 << kCommonRes[i].w << "x" << kCommonRes[i].h
147 << " [" << kCommonRes[i].label << "]\n";
148 }
149 std::cout << " (W) Change Width\n";
150 std::cout << " (H) Change Height\n";
151 std::cout << " (Enter) Go back\n> ";
152
153 std::string sel;
154 std::getline(std::cin, sel);
155 if (sel.empty()) return;
156
157 if (sel.size() == 1 && std::tolower(static_cast<unsigned char>(sel[0])) == 'w') {
158 const int w = readNumberOrDefault<int>(
159 "Enter width", rc.width,
160 [](const int v) { return clampi(v, MIN_W, MAX_W); }
161 );
162 rc.width = w;
163 } else if (sel.size() == 1 && std::tolower(static_cast<unsigned char>(sel[0])) == 'h') {
164 const int h = readNumberOrDefault<int>(
165 "Enter height", rc.height,
166 [](const int v) { return clampi(v, MIN_H, MAX_H); }
167 );
168 rc.height = h;
169 } else {
170 try {
171 if (const int idx = std::stoi(sel); idx >= 1 && idx <= static_cast<int>(std::size(kCommonRes))) {
172 rc.width = kCommonRes[static_cast<size_t>(idx - 1)].w;
173 rc.height = kCommonRes[static_cast<size_t>(idx - 1)].h;
174 }
175 } catch (...) {
176 /* ignore */
177 }
178 }
179 }

420RuntimeConfig promptUserForConfig() {
421 RuntimeConfig rc{};
422 rc.width = 1280;
423 rc.height = 720;
424 rc.sceneMask = 0;
425 rc.enablePostFX = false;
426 rc.fxFilter = 0; // Gaussian
427 rc.gaussRadius = 2;
428 rc.gaussSigma = 1.2f;
429 rc.bilateralRadius = 3;
430 rc.bilateralSigmaSpatial = 2.0f;
431 rc.bilateralSigmaRange = 0.15f;
432 rc.dbgDrawLightSphere = false;
433 rc.dbgDrawLightDir = false;
434 rc.dbgDrawNormals = false;
435 rc.exportFormat = 0; // PPM
436 rc.autoOpenPreview = false;
437 rc.addWatermark = false;
438
439 for (;;) {
440 printHeader(rc);
441 std::string choice;
442 std::getline(std::cin, choice);
443 if (choice.empty()) break;
444
445 if (choice == "1") submenuResolution(rc);
446 else if (choice == "2") submenuScene(rc);
447 else if (choice == "3") submenuDebug(rc);
448 else if (choice == "4") submenuPostFX(rc);
449 else if (choice == "5") submenuExportSettings(rc);
450 }
451
452 rc.width = clampi(rc.width, MIN_W, MAX_W);
453 rc.height = clampi(rc.height, MIN_H, MAX_H);
454 return rc;

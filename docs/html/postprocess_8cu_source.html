52 inline uchar3 packRGB_u8(const float r, const float g, const float b) {
53 auto toU8 = [](const float x) -> unsigned char {
54 const float clamped = (x < 0.0f) ? 0.0f : (x > kU8Max ? kU8Max : x);
55 return static_cast<unsigned char>(std::lround(clamped));
56 };
57 return make_uchar3(toU8(r), toU8(g), toU8(b));
58 }

80 std::vector<float> makeGaussian1D(const int radius, const float sigma) {
81 const int size = 2 * radius + 1;
82 std::vector<float> kernel(size > 0 ? static_cast<size_t>(size) : size_t{1}, 0.0f);
83
84 if (radius <= 0 || sigma <= 0.0f) {
85 kernel[0] = 1.0f;
86 return kernel;
87 }
88
89 const float invTwoSigma2 = 1.0f / (2.0f * sigma * sigma);
90 float sum = 0.0f;
91
92 for (int i = 0; i < size; ++i) {
93 const int x = i - radius;
94 const auto xf = static_cast<float>(x);
95 const float w = std::exp(-(xf * xf) * invTwoSigma2);
96 kernel[static_cast<size_t>(i)] = w;
97 sum += w;
98 }
99 const float invSum = (sum > kSmallEps) ? (1.0f / sum) : 1.0f;
100 for (float &w: kernel) w *= invSum;
101
102 return kernel;
103 }

114 void cpuGaussianRGB(uchar3 *image, const int width, const int height,
115 const int radius, const float sigma) {
116 if (!image || width <= 0 || height <= 0) return;
117
118 const std::vector<float> kernel = makeGaussian1D(radius, sigma);
119 const int kernelSize = static_cast<int>(kernel.size());
120 std::vector<uchar3> scratch(static_cast<size_t>(width) * static_cast<size_t>(height));
121
122 // Horizontal pass: image -> scratch
123 for (int y = 0; y < height; ++y) {
124 const int row = y * width;
125 for (int x = 0; x < width; ++x) {
126 float r = 0.0f, g = 0.0f, b = 0.0f;
127 for (int k = 0; k < kernelSize; ++k) {
128 const int xx = num::clampi(x + (k - radius), 0, width - 1);
129 const uchar3 px = image[row + xx];
130 const float w = kernel[static_cast<size_t>(k)];
131 r += w * static_cast<float>(px.x);
132 g += w * static_cast<float>(px.y);
133 b += w * static_cast<float>(px.z);
134 }
135 scratch[row + x] = packRGB_u8(r, g, b);
136 }
137 }
138
139 // Vertical pass: scratch -> image
140 for (int y = 0; y < height; ++y) {
141 const int row = y * width;
142 for (int x = 0; x < width; ++x) {
143 float r = 0.0f, g = 0.0f, b = 0.0f;
144 for (int k = 0; k < kernelSize; ++k) {
145 const int yy = num::clampi(y + (k - radius), 0, height - 1);
146 const uchar3 px = scratch[static_cast<size_t>(yy) * width + x];
147 const float w = kernel[static_cast<size_t>(k)];
148 r += w * static_cast<float>(px.x);
149 g += w * static_cast<float>(px.y);
150 b += w * static_cast<float>(px.z);
151 }
152 image[row + x] = packRGB_u8(r, g, b);
153 }
154 }

167 void cpuBilateralRGB(uchar3 *image, const int width, const int height,
168 const int radius, const float sigmaSpatial, const float sigmaRange) {
169 if (!image || width <= 0 || height <= 0) return;
170
171 // Snapshot source to avoid feedback within the window
172 const std::vector src(image, image + static_cast<size_t>(width) * static_cast<size_t>(height));
173
174 // Spatial weights ((2R+1)^2 table)
175 const int ksize = 2 * radius + 1;
176 std::vector spatial(static_cast<size_t>(ksize) * static_cast<size_t>(ksize), 1.0f);
177 if (radius > 0 && sigmaSpatial > 0.0f) {
178 const float invTwoSigmaS2 = 1.0f / (2.0f * sigmaSpatial * sigmaSpatial);
179 for (int dy = -radius; dy <= radius; ++dy) {
180 for (int dx = -radius; dx <= radius; ++dx) {
181 const auto d2 = static_cast<float>(dx * dx + dy * dy);
182 spatial[static_cast<size_t>((dy + radius) * ksize + (dx + radius))] =
183 std::exp(-d2 * invTwoSigmaS2);
184 }
185 }
186 }
187
188 // Range LUT for luma deltas 0..255; <=1 means normalized input
189 float range[kU8Range];
190 if (const float sigmaR = (sigmaRange <= 1.0f) ? (sigmaRange * 255.0f) : sigmaRange; sigmaR > 0.0f) {
191 const float invTwoSigmaR2 = 1.0f / (2.0f * sigmaR * sigmaR);
192 for (int d = 0; d < kU8Range; ++d) {
193 const auto df = static_cast<float>(d);
194 range[d] = std::exp(-(df * df) * invTwoSigmaR2);
195 }
196 } else {
197 for (float &d: range) d = 1.0f;
198 }
199
200 // Filter each pixel
201 for (int y = 0; y < height; ++y) {
202 for (int x = 0; x < width; ++x) {
203 const uchar3 center = src[static_cast<size_t>(y) * width + x];
204 const int L0 = luma_u8(center);
205
206 float sumR = 0.0f, sumG = 0.0f, sumB = 0.0f, sumW = 0.0f;
207
208 for (int dy = -radius; dy <= radius; ++dy) {
209 const int yy = num::clampi(y + dy, 0, height - 1);
210 const int row = (dy + radius) * ksize;
211 for (int dx = -radius; dx <= radius; ++dx) {
212 const int xx = num::clampi(x + dx, 0, width - 1);
213 const uchar3 s = src[static_cast<size_t>(yy) * width + xx];
214
215 const float ws = spatial[static_cast<size_t>(row + (dx + radius))];
216 const int dL = std::abs(luma_u8(s) - L0);
217 const float wr = range[dL];
218 const float w = ws * wr;
219
220 sumR += w * static_cast<float>(s.x);
221 sumG += w * static_cast<float>(s.y);
222 sumB += w * static_cast<float>(s.z);
223 sumW += w;
224 }
225 }
226
227 const float invW = (sumW > kSmallEps) ? (1.0f / sumW) : 1.0f;
228 image[static_cast<size_t>(y) * width + x] =
229 packRGB_u8(sumR * invW, sumG * invW, sumB * invW);
230 }
231 }
232 }

236 const Params &p, Timings *t) {
237 const auto t0 = std::chrono::high_resolution_clock::now();
238
239 switch (p.filter) {

272static void uploadGaussianKernel(const int radius, const float sigma) {
273 gGaussianRadius = (radius < 0) ? 0 : (radius > kGaussianMaxRadius ? kGaussianMaxRadius : radius);
274 gGaussianSize = 2 * gGaussianRadius + 1;
275
276 std::vector<float> kernel(static_cast<size_t>(gGaussianSize), 1.0f);
277 if (gGaussianRadius > 0 && sigma > 0.0f) {
278 const float invTwoSigma2 = 1.0f / (2.0f * sigma * sigma);
279 float sum = 0.0f;
280 for (int i = 0; i < gGaussianSize; ++i) {
281 const int x = i - gGaussianRadius;
282 const auto xf = static_cast<float>(x);
283 const float w = std::exp(-(xf * xf) * invTwoSigma2);
284 kernel[static_cast<size_t>(i)] = w;
285 sum += w;
286 }
287 const float invSum = (sum > kSmallEps) ? (1.0f / sum) : 1.0f;
288 for (float &w: kernel) w *= invSum;
289 }
290 CUDA_GUARD(cudaMemcpyToSymbol(cGaussian, kernel.data(),
291 sizeof(float) * static_cast<size_t>(gGaussianSize),
292 0, cudaMemcpyHostToDevice));
293}

378 gSpatialRadius = (radius < 0) ? 0 : (radius > kSpatialMaxRadius ? kSpatialMaxRadius : radius);
379 gSpatialSize = 2 * gSpatialRadius + 1;
380
381 // Spatial ( (2R+1)^2 )
382 const int ksize = (gSpatialSize > 0) ? gSpatialSize : 1;
383 std::vector spatial(static_cast<size_t>(ksize) * static_cast<size_t>(ksize), 1.0f);
384 if (gSpatialRadius > 0 && sigmaSpatial > 0.0f) {
385 const float invTwoSigmaS2 = 1.0f / (2.0f * sigmaSpatial * sigmaSpatial);
386 for (int dy = -gSpatialRadius; dy <= gSpatialRadius; ++dy) {

388 const auto d2 = static_cast<float>(dx * dx + dy * dy);
389 spatial[static_cast<size_t>((dy + gSpatialRadius) * ksize + (dx + gSpatialRadius))] =

392 }
393 }

397 // Range (0..255); parameter is normalized (0..1) or absolute (>=1)
398 float range[kU8Range];

400 const float invTwoSigmaR2 = 1.0f / (2.0f * sigmaR * sigmaR);
401 for (int d = 0; d < kU8Range; ++d) {
402 const auto df = static_cast<float>(d);
403 range[d] = std::exp(-(df * df) * invTwoSigmaR2);
404 }
405 } else {
406 for (float &d: range) d = 1.0f;
407 }

423 const int width, const int height, const int radius, const int ksize) {
424 const int x = static_cast<int>(blockIdx.x) * static_cast<int>(blockDim.x) + static_cast<int>(threadIdx.x);

426 if (x >= width || y >= height) return;

428 const int idxCenter = y * width + x;

433

436 const int row = (dy + radius) * ksize;

442 const int dL = abs(luma_u8(make_uchar3(s.x, s.y, s.z)) - L0);
443 const float wr = cRange[dL];
444 const float w = ws * wr;
445
446 sumR += w * static_cast<float>(s.x);
447 sumG += w * static_cast<float>(s.y);
448 sumB += w * static_cast<float>(s.z);
449 sumW += w;
450 }
451 }
452

454 const float invW = 1.0f / sumW;
455 out[idxCenter] = make_uchar4(
456 static_cast<unsigned char>(fminf(kU8Max, sumR * invW + 0.5f)),
457 static_cast<unsigned char>(fminf(kU8Max, sumG * invW + 0.5f)),
458 static_cast<unsigned char>(fminf(kU8Max, sumB * invW + 0.5f)),
459 center.w
460 );
461 } else {

466void PostFX::applyGPU(uchar4 *&d_img, const int width, const int height, const Params &p, Timings *t,
467 cudaStream_t stream) {
468 if (p.filter == Filter::None) {
469 if (t) t->ms = 0.0f;
470 return;
471 }
472
473 cudaEvent_t start{}, stop{};
474 CUDA_GUARD(cudaEventCreate(&start));
475 CUDA_GUARD(cudaEventCreate(&stop));
476 CUDA_GUARD(cudaEventRecord(start, stream));
477
478 constexpr dim3 block(kBlockDimX, kBlockDimY);
479 const dim3 grid((width + block.x - 1) / block.x,
480 (height + block.y - 1) / block.y);
481
482 if (p.filter == Filter::Gaussian) {
483 const int radius = (p.gaussianRadius > kGaussianMaxRadius)
484 ? kGaussianMaxRadius
485 : (p.gaussianRadius < 0 ? 0 : p.gaussianRadius);
486 uploadGaussianKernel(radius, p.gaussianSigma);
487
488 uchar4 *d_tmp = nullptr;
489 CUDA_GUARD(cudaMallocAsync(&d_tmp,
490 static_cast<size_t>(width) * static_cast<size_t>(height) * sizeof(uchar4), stream));
491
492 kGaussianH<<<grid, block, 0, stream>>>(d_img, d_tmp, width, height, gGaussianRadius);
493 CUDA_GUARD(cudaGetLastError());
494
495 kGaussianV<<<grid, block, 0, stream>>>(d_tmp, d_img, width, height, gGaussianRadius);
496 CUDA_GUARD(cudaGetLastError());
497
498 CUDA_GUARD(cudaFreeAsync(d_tmp, stream));
499 } else {
500 // Bilateral
501 const int radius = (p.bilateralRadius > kSpatialMaxRadius)
502 ? kSpatialMaxRadius
503 : (p.bilateralRadius < 0 ? 0 : p.bilateralRadius);
504 uploadBilateralTables(radius, p.bilateralSigmaSpatial, p.bilateralSigmaRange);
505
506 uchar4 *d_tmp = nullptr;
507 CUDA_GUARD(cudaMallocAsync(&d_tmp,
508 static_cast<size_t>(width) * static_cast<size_t>(height) * sizeof(uchar4), stream));
509
510 kBilateralNaive<<<grid, block, 0, stream>>>(d_img, d_tmp, width, height, gSpatialRadius, gSpatialSize);
511 CUDA_GUARD(cudaGetLastError());
512
513 std::swap(d_img, d_tmp);
514
515 CUDA_GUARD(cudaFreeAsync(d_tmp, stream));
516 }
517
518 CUDA_GUARD(cudaEventRecord(stop, stream));
519 CUDA_GUARD(cudaEventSynchronize(stop));
520 if (t) { CUDA_GUARD(cudaEventElapsedTime(&t->ms, start, stop)); }
521 CUDA_GUARD(cudaEventDestroy(start));
522 CUDA_GUARD(cudaEventDestroy(stop));
523}

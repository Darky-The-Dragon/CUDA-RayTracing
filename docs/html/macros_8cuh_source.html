77#define CUDA_GUARD(expr) \
78 do { \
79 cudaError_t _e = (expr); \
80 if (_e != cudaSuccess) { \
81 std::fprintf(stderr, "[CUDA] %s failed at %s:%d : %s\n", \
82 #expr, __FILE__, __LINE__, cudaGetErrorString(_e)); \
83 std::fflush(stderr); \
84 std::abort(); \
85 } \
86 } while (0)

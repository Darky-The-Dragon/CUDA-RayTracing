102HD FINL inline bool intersectsSphere(const Ray &ray, const Vec3 &center, float radius) {
103 Sphere s;
104 s.center = center;
105 s.radius = radius;
106
107 float t0, t1;
108 if (!s.intersectBoth(ray.origin, ray.direction, t0, t1)) return false;
109
110 // True if at least one hit is in front of the origin.
111 return (t0 > num::kEps()) || (t1 > num::kEps());
112}

149HD FINL inline bool renderLightDirectionRay(const Ray &ray, const Light &light, uchar3 &outColor) {
150#if DEBUG_DRAW_LIGHT_DIRECTION
151 if (!dbgLightDir()) return false;
152 if (light.type == POINT) return false;
153
154 const Vec3 dirNorm = light.direction.normalize();
155 const Vec3 lineMid = light.position + dirNorm * (0.5f * kArrowBodyLength);
156
157 if (intersectsSphere(ray, lineMid, kArrowBodyRadius)) {
158 outColor = make_uchar3(255, 0, 255);
159 return true;
160 }
161 return false;
162#else
163 (void) ray; (void) light; (void) outColor;
164 return false;
165#endif
166}

112HD inline void sampleLight(const Light &light, const Vec3 &P, Vec3 &L, float &attenuation, float &distToLight) {
113 if (light.type == POINT || light.type == SPOT) {
114 const Vec3 d = light.position - P;
115 distToLight = d.length();
116 L = (distToLight > 0.0f) ? (d / distToLight) : Vec3(0.0f);
117 attenuation = 1.0f / fmaxf(distToLight * distToLight, num::kMinInvDistanceSq());
118 return;
119 }
120 // DIRECTIONAL
121 L = (-light.direction).normalize();
122 attenuation = 1.0f;
123 distToLight = num::kHuge();
124}

139HD inline bool isOccluded(const Vec3 &P, const Vec3 &N, const Vec3 &L, const float maxDist, const SceneGeom &G) {
140 const Ray r(P + N * num::kShadowRayBias(), L);
141
142 for (int i = 0; i < G.numQuads; ++i) {
143 if (float t = 0.0f; G.quads[i].intersect(r, t) && t > 0.0f && t < (maxDist - num::kShadowEndPad()))
144 return true;
145 }
146 for (int i = 0; i < G.numSpheres; ++i) {
147 if (float t = 0.0f; G.spheres[i].intersect(r.origin, r.direction, t) &&
148 t > 0.0f && t < (maxDist - num::kShadowEndPad()))
149 return true;
150 }
151 return false;
152}

200HD inline void sampleDisk(const float u1, const float u2, const float radius, float &dx, float &dy) {
201 const float a = 2.0f * u1 - 1.0f, b = 2.0f * u2 - 1.0f;
202 if (a == 0.0f && b == 0.0f) {
203 dx = dy = 0.0f;
204 return;
205 }
206 float r, phi;
207 if (fabsf(a) > fabsf(b)) {
208 r = a;
209 phi = (num::kPi() * 0.25f) * (b / a);
210 } else {
211 r = b;
212 phi = (num::kPi() * 0.5f) - (num::kPi() * 0.25f) * (a / b);
213 }
214 dx = radius * r * cosf(phi);
215 dy = radius * r * sinf(phi);
216}

233HD inline float softShadowVisibility(const Vec3 &P, const Vec3 &N, const Light &light, const SceneGeom &G,
234 uint32_t seed, const int samples, const bool useBentShadows) {
235 // Hard-shadow path
236 if (light.radius <= 0.0f || samples <= 1) {
237 Vec3 L;
238 float att, dist;
239 sampleLight(light, P, L, att, dist);
240 if (!useBentShadows) return isOccluded(P, N, L, dist, G) ? 0.0f : 1.0f;
241
242 const Vec3 samplePos = (light.type == DIRECTIONAL)
243 ? (P + (-light.direction).normalize() * num::kDirectionalShadowDistance())
244 : light.position;
245 return visibilityBentOneRefractor(P, N, samplePos, G);
246 }
247
248 // Area light: stratified samples on emitter disk
249 const Vec3 toL = light.position - P;
250 const float d = toL.length();
251 const Vec3 Ldir = (d > 0.0f) ? (toL / d) : Vec3(0, 1, 0);
252
253 Vec3 t, b;
254 onb_from_n(Ldir, t, b);
255 const int grid = (samples > 1) ? static_cast<int>(ceilf(sqrtf(static_cast<float>(samples)))) : 0;
256 const float inv = (grid > 0) ? (1.0f / static_cast<float>(grid)) : 0.0f;
257
258 int visible = 0, taken = 0;
259 for (int gy = 0; gy < grid && taken < samples; ++gy) {
260 for (int gx = 0; gx < grid && taken < samples; ++gx) {
261 const float u = (static_cast<float>(gx) + rand01(seed)) * inv;
262 const float v = (static_cast<float>(gy) + rand01(seed)) * inv;
263
264 float dx, dy;
265 sampleDisk(u, v, light.radius, dx, dy);
266 const Vec3 sampleP = light.position + t * dx + b * dy;
267
268 const bool occ = (!useBentShadows)
269 ? isOccluded(P, N, (sampleP - P).normalize(), (sampleP - P).length(), G)
270 : (visibilityBentOneRefractor(P, N, sampleP, G) < 0.5f);
271
272 if (!occ) ++visible;
273 ++taken;
274 }
275 }
276 return (samples > 0) ? static_cast<float>(visible) / static_cast<float>(samples) : 0.0f;
277}

294HD inline Vec3 shadeLambertUnified(const Hit &h, const Light &light, const SceneGeom &G, const uint32_t seed,
295 const int samples, const Vec3 &ambient = shade_defaults::ambient(),
296 const bool useBentShadows = false) {
297 if (!h.hit) return Vec3(0.0f);
298
299 Vec3 L;
300 float att, dist;
301 sampleLight(light, h.P, L, att, dist);
302 const float ndotl = fmaxf(0.0f, h.N.dot(L));
303 const float vis = softShadowVisibility(h.P, h.N, light, G, seed, samples, useBentShadows);
304
305 const Vec3 base = toFloat3(h.mat.color);
306 const Vec3 Li = toFloat3(light.color) * light.intensity * att;
307
308 return gammaEncode(base * (ndotl * Li * vis) + base * ambient);
309}

386HD inline bool refractDir(const Vec3 &I, const Vec3 &N, const float eta, Vec3 &T_out) {
387 const float cosI = -fmaxf(-1.0f, fminf(1.0f, I.dot(N)));
388 const float sin2T = eta * eta * (1.0f - cosI * cosI);
389 if (sin2T > 1.0f) return false;
390 const float cosT = sqrtf(fmaxf(0.0f, 1.0f - sin2T));
391 T_out = (I * eta + N * (eta * cosI - cosT)).normalize();
392 return true;
393}

401HD inline float fresnelSchlick(const float cosTheta, const float ior1, const float ior2) {
402 float R0 = (ior1 - ior2) / (ior1 + ior2);
403 R0 *= R0;
404 const float m = 1.0f - cosTheta;
405 return R0 + (1.0f - R0) * (m * m * m * m * m);
406}

419HD inline bool traceClosest(const Ray &ray, const SceneGeom &G, Hit &outHit) {
420 outHit.hit = false;
421 outHit.t = num::kHuge();
422
423 for (int i = 0; i < G.numQuads; ++i) {
424 if (float tHit = 0.0f; G.quads[i].intersect(ray, tHit) &&
425 tHit > num::kHitMinT() && tHit < outHit.t) {
426 outHit.hit = true;
427 outHit.t = tHit;
428 outHit.P = ray.at(tHit);
429 outHit.N = G.quads[i].normal;
430 outHit.mat = G.quads[i].material;
431 }
432 }
433 for (int i = 0; i < G.numSpheres; ++i) {
434 if (float tHit = 0.0f; G.spheres[i].intersect(ray.origin, ray.direction, tHit) &&
435 tHit > num::kHitMinT() && tHit < outHit.t) {
436 outHit.hit = true;
437 outHit.t = tHit;
438 outHit.P = ray.at(tHit);
439 outHit.N = (outHit.P - G.spheres[i].center).normalize();
440 outHit.mat = G.spheres[i].material;
441 }
442 }
443 return outHit.hit;
444}

463HD inline Vec3 shadeSurface(const Hit &primaryHit, const Ray &viewRay, const Light &light, const SceneGeom &G,
464 const uint32_t seed, const int maxDepth, const int softShadowSamples, const Vec3 &bgLinear,
465 const bool useBentShadows) {
466 if (!primaryHit.hit) return bgLinear;
467
468 const Material &mat = primaryHit.mat;
469 const Vec3 base = toFloat3(mat.color);
470
471 // Diffuse
472 if (mat.type == DIFFUSE) {
473 return shadeLambertUnified(primaryHit, light, G, seed, softShadowSamples,
474 shade_defaults::ambient(), useBentShadows);
475 }
476
477 // Depth exhausted → tiny ambient fallback
478 if (maxDepth <= 0) return gammaEncode(base * shade_defaults::ambient());
479
480 const Vec3 N = primaryHit.N;
481 const Vec3 V = (-viewRay.direction).normalize();
482
483 auto traceAndShade = [&](const Ray &r) -> Vec3 {
484 if (Hit h2{}; traceClosest(r, G, h2)) {
485 return shadeSurface(h2, r, light, G, wanghash(seed), maxDepth - 1,
486 softShadowSamples, bgLinear, useBentShadows);
487 }
488 return bgLinear;
489 };
490
491 // Mirror
492 if (mat.type == REFLECTIVE) {
493 const Vec3 Rdir = reflectDir(-V, N).normalize();
494 const Vec3 Rc = traceAndShade(Ray(primaryHit.P + N * num::kShadowRayBias(), Rdir));
495 const float glossy = fmaxf(0.0f, 1.0f - mat.roughness);
496 return gammaEncode(Rc * glossy + base * (1.0f - glossy));
497 }
498
499 // Glass
500 if (mat.type == REFRACTIVE) {
501 constexpr float iorAir = 1.0f;
502 float eta = iorAir / fmaxf(num::kFloatEps(), mat.ior);
503 Vec3 Nf = N;
504 float cosI = V.dot(N);
505 const bool entering = (cosI >= 0.0f);
506 if (!entering) {
507 eta = mat.ior / iorAir;
508 Nf = -N;
509 cosI = -cosI;
510 }
511
512 const Vec3 Rdir = reflectDir(-V, Nf).normalize();
513 const Vec3 Rc = traceAndShade(Ray(primaryHit.P + Nf * num::kShadowRayBias(), Rdir));
514
515 Vec3 Tc(0.0f);
516 float Kr = fresnelSchlick(cosI, entering ? iorAir : mat.ior,
517 entering ? mat.ior : iorAir);
518
519 if (Vec3 Tdir; refractDir(-V, Nf, eta, Tdir)) {
520 Tc = traceAndShade(Ray(primaryHit.P - Nf * num::kShadowRayBias(), Tdir));
521 } else {
522 Kr = 1.0f; // TIR
523 }
524
525 const float kt = (1.0f - Kr) * (1.0f - fminf(1.0f, mat.opacity));
526 Vec3 linear = Rc * Kr + Tc * kt;
527 linear = linear * base; // optional tint
528 return gammaEncode(linear);
529 }
530
531 // Fallback
532 return gammaEncode(base);
533}

599 HD inline int traceSphereEntryExit(const Ray &r, const SceneGeom &G, const float maxDist, float &tEnter,
600 float &tExit, Vec3 &normalAtEnter, Material &matOut) {
601 int sphIdx = -1;
602 float tNear = maxDist;
603
604 for (int i = 0; i < G.numSpheres; ++i) {
605 float t0 = 0.0f, t1 = 0.0f;
606 if (G.spheres[i].intersectBoth(r.origin, r.direction, t0, t1)) {
607 if (t1 > 0.0f && t0 < maxDist && t0 < tNear) {
608 tNear = t0;
609 sphIdx = i;
610 tEnter = t0;
611 tExit = t1;
612 }
613 }
614 }
615 if (sphIdx >= 0) {
616 const Vec3 Penter = r.at(tEnter);
617 normalAtEnter = (Penter - G.spheres[sphIdx].center).normalize();
618 matOut = G.spheres[sphIdx].material;
619 }
620 return sphIdx;
621 }

632HD inline float visibilityBentOneRefractor(const Vec3 &P, const Vec3 &N, const Vec3 &lightSample, const SceneGeom &G) {
633 using namespace detail;
634
635 const Vec3 toSample = lightSample - P;
636 const float fullDist = toSample.length();
637 if (fullDist <= 0.0f) return 1.0f;
638
639 const Vec3 dir0 = toSample / fullDist;
640 Ray r0(P + N * epsOrigin(), dir0);
641
642 // Quick classification along the straight segment toward the light.
643 {
644 Hit h{};
645 const HitKind k = traceClosestKind(r0, G, fullDist, h);
646 if (!h.hit) return 1.0f;
647
648 if (k == HitKind::Quad) {
649 if (h.mat.type == REFRACTIVE) {
650 constexpr float kRefractiveQuadTransmit = 0.35f; // tweakable
651 return kRefractiveQuadTransmit;
652 }
653 return 0.0f; // opaque/reflective quads fully occlude
654 }
655 if (k == HitKind::Sphere &&
656 (h.mat.type != REFRACTIVE || h.mat.opacity >= 1.0f)) {
657 return 0.0f;
658 }
659 }
660
661 // Bent path through one refractive sphere
662 float tEnter = farDist(), tExit = farDist();
663 Vec3 Nenter(0.0f);
664 Material glass{};
665 const int sphIdx = traceSphereEntryExit(r0, G, fullDist, tEnter, tExit, Nenter, glass);
666 if (sphIdx < 0) return 1.0f;
667 if (glass.type != REFRACTIVE) return 0.0f;
668
669 constexpr float iorAir = 1.0f;
670 const float iorGlass = fmaxf(1e-3f, glass.ior);
671 const float etaIn = iorAir / iorGlass;
672 const float etaOut = iorGlass / iorAir;
673
674 Vec3 dirInside;
675 if (!refractDir(r0.direction, Nenter, etaIn, dirInside)) return 0.0f; // TIR on entry
676
677 const Vec3 Pexit = r0.at(tExit);
678 const Vec3 Nexit = (Pexit - G.spheres[sphIdx].center).normalize();
679
680 if (Vec3 dirOut; !refractDir(dirInside, -Nexit, etaOut, dirOut)) return 0.0f; // TIR on exit
681
682 const Vec3 remain = lightSample - (Pexit + Nexit * epsExit());
683 const float restLen = remain.length();
684 if (restLen <= 0.0f) return 1.0f;
685
686 Ray r1(Pexit + Nexit * epsExit(), remain / restLen);
687
688 // Any hit along the rest path blocks
689 for (int i = 0; i < G.numQuads; ++i) {
690 if (float t; G.quads[i].intersect(r1, t) && t > 0.0f && t < restLen) return 0.0f;
691 }
692 for (int i = 0; i < G.numSpheres; ++i) {
693 if (float t; G.spheres[i].intersect(r1.origin, r1.direction, t) && t > 0.0f && t < restLen) return 0.0f;
694 }
695 return 1.0f;
696}

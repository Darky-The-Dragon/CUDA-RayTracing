49 double gb_per_s_for_bytes(const std::size_t bytes, const double ms) {
50 if (ms <= 0.0) return 0.0;
51 const double gb = static_cast<double>(bytes) / 1e9; // decimal GB
52 return gb / (ms / 1e3);

65GpuInfo queryGpuInfo() {
66 GpuInfo gi{};
67 int dev = 0;
68 CUDA_GUARD(cudaGetDevice(&dev));
69
70 cudaDeviceProp p{};
71 CUDA_GUARD(cudaGetDeviceProperties(&p, dev));
72
73 gi.name = p.name ? p.name : "Unknown";
74 gi.major = p.major;
75 gi.minor = p.minor;
76 gi.sms = p.multiProcessorCount;
77 gi.clockKHz = p.clockRate;
78 gi.memClockKHz = p.memoryClockRate;
79 gi.memBusBits = p.memoryBusWidth;
80 gi.globalMemBytes = p.totalGlobalMem;
81 return gi;
82}

84void printRunSummary(const GpuInfo &gi, const RunStats &rs) {
85 const double mpix = static_cast<double>(rs.pixels) / 1e6;
86 const double gpuMPixPerS = (rs.gpuPrimaryMs > 0.0) ? (mpix / (rs.gpuPrimaryMs / 1e3)) : 0.0;
87 const double cpuMPixPerS = (rs.cpuPrimaryMs > 0.0) ? (mpix / (rs.cpuPrimaryMs / 1e3)) : 0.0;
88
89 std::size_t postfxBytes = 0;
90 if (rs.filter == PostFX::Filter::Gaussian) postfxBytes = rs.imageBytes * 4ull; // H & V (R+W)
91 else if (rs.filter == PostFX::Filter::Bilateral) postfxBytes = rs.imageBytes * 2ull; // 1 pass (R+W)
92
93 const double gpuFxGBs = gb_per_s_for_bytes(postfxBytes, rs.gpuFxMs);
94 const double gpuTotalMs = rs.gpuPrimaryMs + rs.gpuFxMs;
95 const double cpuTotalMs = rs.cpuPrimaryMs + rs.cpuFxMs;
96
97 std::cout << "\n==================== Run Summary ====================\n";
98 std::cout << " GPU : " << gi.name
99 << " | CC " << gi.major << "." << gi.minor
100 << " | SMs " << gi.sms
101 << " | GlobalMem " << (gi.globalMemBytes / (1024 * 1024)) << " MiB\n";
102 std::cout << " Grid/Block : (" << rs.grid.x << "x" << rs.grid.y
103 << ") / (" << rs.block.x << "x" << rs.block.y << ")\n";
104 std::cout << " Image : " << rs.width << " x " << rs.height
105 << " (" << std::fixed << std::setprecision(2) << mpix << " MPix)\n";
106 std::cout << " Scene : " << (rs.sceneLabel.empty() ? "None" : rs.sceneLabel) << "\n";
107 std::cout << " Filter : " << fxName(rs.filter)
108 << " | Settings : " << (rs.fxSettingsLabel.empty() ? "DEFAULT" : rs.fxSettingsLabel) << "\n";
109
110 std::cout << " GPU primary : " << std::setprecision(3) << rs.gpuPrimaryMs << " ms ("
111 << std::setprecision(2) << gpuMPixPerS << " MPix/s)\n";
112 if (rs.filter != PostFX::Filter::None) {
113 std::cout << " GPU post-FX : " << std::setprecision(3) << rs.gpuFxMs << " ms";
114 if (rs.gpuFxMs > 0.0) {
115 std::cout << " | logical BW ~ " << std::setprecision(2) << gpuFxGBs << " GB/s";
116 }
117 std::cout << "\n";
118 }
119 std::cout << " GPU total : " << std::setprecision(3) << gpuTotalMs << " ms\n";
120 std::cout << " CPU primary : " << std::setprecision(3) << rs.cpuPrimaryMs << " ms ("
121 << std::setprecision(2) << cpuMPixPerS << " MPix/s)\n";
122 if (rs.filter != PostFX::Filter::None) {
123 std::cout << " CPU post-FX : " << std::setprecision(3) << rs.cpuFxMs << " ms\n";
124 }
125 std::cout << " CPU total : " << std::setprecision(3) << cpuTotalMs << " ms\n";
126 std::cout << "=====================================================\n";
127}

129void appendTimingsCSV(const std::string &outDir, const GpuInfo &gi, const RunStats &rs) {
130 // Ensure logs dir exists
131 const fs::path csvPath = fs::path(outDir) / "logs" / "timings.csv";
132 fs::create_directories(csvPath.parent_path());
133 const bool needHeader = !fs::exists(csvPath);
134
135 std::ofstream f(csvPath, std::ios::app);
136 if (!f) {
137 std::cerr << "[ERROR] Could not open timings.csv for writing: " << csvPath << "\n";
138 return;
139 }
140
141 // Header: append new fields at the end for backward compatibility
142 if (needHeader) {
143 f << "Datetime,Gpu,CC,SMS,Resolution,Scene,Filter,PostFX Settings,"
144 "GPU Primary (ms),GPU PostFX (ms),GPU Total (ms),"
145 "CPU Primary (ms),CPU PostFX (ms),CPU Total (ms),"
146 "MPx,GPU MPx (s),CPU MPx (s),PostFX (bytes),GPU PostFX (GB/s),"
147 "GPU Grid X,GPU Grid Y,GPU Block X,GPU Block Y,"
148 "Seed,Repeats,RunIndex,CRC_GPU_RAW,CRC_CPU_RAW\n";
149 }
150
151 // Timestamp (local)
152 const std::time_t t = std::time(nullptr);
153 char buf[32];
154#if defined(_WIN32)
155 std::tm tm_buf{};
156 localtime_s(&tm_buf, &t);
157 std::strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", &tm_buf);
158#else
159 std::tm tm_buf = *std::localtime(&t);
160 std::strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", &tm_buf);
161#endif
162
163 const std::string cc = std::to_string(gi.major) + "." + std::to_string(gi.minor);
164 const std::string res = std::to_string(rs.width) + "x" + std::to_string(rs.height);
165 const double mpix = static_cast<double>(rs.pixels) / 1e6;
166 const double gpuMPixPerS = (rs.gpuPrimaryMs > 0.0) ? (mpix / (rs.gpuPrimaryMs / 1e3)) : 0.0;
167 const double cpuMPixPerS = (rs.cpuPrimaryMs > 0.0) ? (mpix / (rs.cpuPrimaryMs / 1e3)) : 0.0;
168
169 std::size_t postfxBytes = 0;
170 if (rs.filter == PostFX::Filter::Gaussian) postfxBytes = rs.imageBytes * 4ull;
171 else if (rs.filter == PostFX::Filter::Bilateral) postfxBytes = rs.imageBytes * 2ull;
172
173 const double gpuFxGBs = (rs.gpuFxMs > 0.0)
174 ? (static_cast<double>(postfxBytes) / 1e9) / (rs.gpuFxMs / 1e3)
175 : 0.0;
176
177 const double gpuTotalMs = rs.gpuPrimaryMs + rs.gpuFxMs;
178 const double cpuTotalMs = rs.cpuPrimaryMs + rs.cpuFxMs;
179 const bool fxOff = (rs.filter == PostFX::Filter::None);
180
181 // Row start
182 f << buf << ','
183 << '"' << gi.name << '"' << ','
184 << cc << ','
185 << gi.sms << ','
186 << res << ','
187 << '"' << (rs.sceneLabel.empty() ? "None" : rs.sceneLabel) << '"' << ','
188 << (rs.filter == PostFX::Filter::None
189 ? "Off"
190 : (rs.filter == PostFX::Filter::Gaussian ? "Gaussian" : "Bilateral")) << ','
191 << '"' << (rs.fxSettingsLabel.empty() ? "DEFAULT" : rs.fxSettingsLabel) << '"' << ',';
192
193 // GPU Primary (ms)
194 f << std::fixed << std::setprecision(3) << rs.gpuPrimaryMs << ',';
195
196 // GPU PostFX (ms)
197 if (fxOff) f << "NaN,";
198 else f << std::fixed << std::setprecision(3) << rs.gpuFxMs << ',';
199
200 // GPU Total (ms)
201 f << std::fixed << std::setprecision(3) << gpuTotalMs << ',';
202
203 // CPU Primary (ms)
204 f << std::fixed << std::setprecision(3) << rs.cpuPrimaryMs << ',';
205
206 // CPU PostFX (ms)
207 if (fxOff) f << "NaN,";
208 else f << std::fixed << std::setprecision(3) << rs.cpuFxMs << ',';
209
210 // CPU Total (ms)
211 f << std::fixed << std::setprecision(3) << cpuTotalMs << ',';
212
213 // MPx, GPU MPx (s), CPU MPx (s)
214 f << std::setprecision(2) << mpix << ','
215 << std::setprecision(2) << gpuMPixPerS << ','
216 << std::setprecision(2) << cpuMPixPerS << ',';
217
218 // PostFX (bytes)
219 if (fxOff) f << 0 << ',';
220 else f << static_cast<long long>(postfxBytes) << ',';
221
222 // GPU PostFX (GB/s)
223 if (fxOff) f << "NaN,";
224 else f << std::fixed << std::setprecision(3) << gpuFxGBs << ',';
225
226 // Grid/Block
227 f << rs.grid.x << ',' << rs.grid.y << ','
228 << rs.block.x << ',' << rs.block.y << ',';
229
230 // New fields (append at end): Seed,Repeats,RunIndex,CRC_GPU_RAW,CRC_CPU_RAW
231 f << rs.seed << ','
232 << rs.repeats << ','
233 << rs.runIndex << ',';
234
235 // CRCs as 8-hex (0xXXXXXXXX). Preserve float flags.
236 const auto old_flags = f.flags();
237 f << "0x" << std::hex << std::uppercase << rs.gpuCRC << ','
238 << "0x" << std::hex << std::uppercase << rs.cpuCRC << '\n';
239 f.flags(old_flags);

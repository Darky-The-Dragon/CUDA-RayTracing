44HD inline Ray generatePrimaryRay(const Camera &cam, const unsigned int x, const unsigned int y, const int width,
45 const int height) {
46 // Basic guards / scalars
47 const auto w = static_cast<float>(width);
48 const auto h = static_cast<float>(height);
49 const float aspect = (h > 0.0f) ? (w / h) : 1.0f;
50 const float fov_rad = num::deg2rad(cam.fov_deg);
51 const float half_tan = tanf(0.5f * fov_rad);
52
53 // NDC in [-1, 1] using pixel centers; bottom-left origin.
54 float ndcX = ((static_cast<float>(x) + 0.5f) / w) * 2.0f - 1.0f;
55 float ndcY = ((static_cast<float>(y) + 0.5f) / h) * 2.0f - 1.0f;
56
57 // For top-left origin, uncomment:
58 // ndcY = -ndcY;
59
60 // Apply aspect and FOV scaling
61 ndcX *= aspect * half_tan;
62 ndcY *= half_tan;
63
64 // Orthonormal basis from forward/up (with safe up fallback)
65 const Vec3 F = cam.forward.normalize();
66
67 Vec3 up = cam.up.normalize();
68 if (fabsf(F.dot(up)) > 0.999f) {
69 // nearly collinear
70 up = Vec3(0.0f, 1.0f, 0.0f);
71 if (fabsf(F.dot(up)) > 0.999f) up = Vec3(0.0f, 0.0f, 1.0f); // second fallback
72 }
73
74 const Vec3 R = F.cross(up).normalize(); // Right
75 const Vec3 U = R.cross(F).normalize(); // Corrected up
76
77 const Vec3 dir = (F + R * ndcX + U * ndcY).normalize();
78 return Ray{cam.position, dir};
79}

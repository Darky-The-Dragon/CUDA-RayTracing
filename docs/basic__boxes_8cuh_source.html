33HD FINL inline Vec3 rotY(const Vec3 &p, const float yawRad) {
34 const float c = cosf(yawRad), s = sinf(yawRad);
35 return Vec3(c * p.x + s * p.z, p.y, -s * p.x + c * p.z);

58HD FINL inline void addBoxQuads(Quad *quads, int &count, const int cap, const Vec3 &center, const float width,
59 const float height, const float depth, const Material &m, const float rotationDegY) {
60 // Require full room for 6 faces — avoid partial geometry.
61 if (count + 6 > cap) return;
62
63 const float hx = 0.5f * width;
64 const float hy = 0.5f * height;
65 const float hz = 0.5f * depth;
66
67 // Build in local box space around center, then rotate & translate.
68 const float x0 = -hx, x1 = hx;
69 const float y0 = -hy, y1 = hy;
70 const float z0 = -hz, z1 = hz;
71
72 const float X = (x1 - x0), Y = (y1 - y0), Z = (z1 - z0);
73 const float yaw = deg2rad(rotationDegY);
74
75 auto push = [&](const Vec3 &pLocal, const Vec3 &uLocal, const Vec3 &vLocal) {
76 Quad q;
77 // Rotate local vectors around +Y, then translate origin to world center.
78 q.position = center + rotY(pLocal, yaw);
79 q.spanU = rotY(uLocal, yaw);
80 q.spanV = rotY(vLocal, yaw);
81 q.material = m;
82 q.normal = (q.spanU.cross(q.spanV)).normalize(); // outward by RH rule
83 quads[count++] = q;
84 };
85
86 // Faces (outward-facing), vetted configuration, expressed in LOCAL space
87 // +X
88 push(Vec3(x1, y0, z0), Vec3(0, Y, 0), Vec3(0, 0, Z));
89 // -X
90 push(Vec3(x0, y0, z1), Vec3(0, Y, 0), Vec3(0, 0, -Z));
91 // +Y
92 push(Vec3(x0, y1, z0), Vec3(0, 0, Z), Vec3(X, 0, 0));
93 // -Y
94 push(Vec3(x0, y0, z1), Vec3(0, 0, -Z), Vec3(X, 0, 0));
95 // +Z
96 push(Vec3(x0, y0, z1), Vec3(X, 0, 0), Vec3(0, Y, 0));
97 // -Z
98 push(Vec3(x1, y0, z0), Vec3(-X, 0, 0), Vec3(0, Y, 0));
99}
